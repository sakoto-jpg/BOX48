```python
import os
import time
import re
import yt_dlp
from typing import List, Dict
from nfo_builder import NfoBuilder
from downloader import BiliDownloader

class Snh48Scanner:
    def __init__(self, config_path: str, cookie_path: str, base_download_path: str):
        self.config_path = config_path
        self.cookie_path = cookie_path
        self.nfo_builder = NfoBuilder(config_path, base_download_path)
        self.downloader = BiliDownloader(cookie_path)
        
        self.history_file = os.path.join(os.path.dirname(config_path), "download_history.txt")
        self._ensure_history_file()
        
        self.active_tasks = {}

    def _ensure_history_file(self):
        if not os.path.exists(self.history_file):
            with open(self.history_file, 'w') as f:
                f.write("")

    def _is_downloaded(self, bvid: str) -> bool:
        with open(self.history_file, 'r') as f:
            return bvid in f.read()

    def _mark_downloaded(self, bvid: str):
        with open(self.history_file, 'a') as f:
            f.write(f"{bvid}\n")

    def _match_team(self, title: str) -> str:
        for team_name, team_data in self.nfo_builder.roster.items():
            keywords = team_data.get("keywords", [])
            for kw in keywords:
                if kw in title or kw.replace(" ", "") in title.replace(" ", ""):
                    return team_name
        return None

    def _create_hook(self, bvid):
        def hook(d):
            if d['status'] == 'downloading':
                percent = d.get('_percent_str', '0%')
                clean_percent = re.sub(r'\x1b\[[0-9;]*m', '', percent).strip()
                if bvid in self.active_tasks:
                    self.active_tasks[bvid]['progress'] = clean_percent
                    self.active_tasks[bvid]['status'] = "视频分片下载中..."
            elif d['status'] == 'finished':
                if bvid in self.active_tasks:
                    self.active_tasks[bvid]['progress'] = '100%'
                    self.active_tasks[bvid]['status'] = "合并音视频流..."
        return hook

    def _process_video_info(self, bvid: str, title: str, video_info: dict, target_team: str, date_str: str, year: str, formatted_date: str):
        parts = video_info.get('entries') if 'entries' in video_info else [video_info]
        if bvid in self.active_tasks:
            self.active_tasks[bvid]['p_total'] = len(parts)

        success_count = 0
        for index, part in enumerate(parts):
            p_number = index + 1
            if bvid in self.active_tasks:
                self.active_tasks[bvid]['p_current'] = p_number

            part_title = part.get('title')
            plot = part.get('description', '')

            # 这里使用了 B 站返回的真实标题 (perf_title=title) 去建立文件夹
            output_path = self.nfo_builder.build_metadata(
                team_name=target_team, year=year, date_str=formatted_date,
                p_number=str(p_number), part_title=part_title, plot=plot,
                perf_title=title
            )
            
            success = self.downloader.download_part(bvid, p_number, output_path, progress_callback=self._create_hook(bvid))
            if success:
                success_count += 1

        # 核心：只有真正下载成功了，才把它写进历史记录！
        if success_count > 0:
            self._mark_downloaded(bvid)
            print(f"🎉 公演入库完成: {title}")
            if bvid in self.active_tasks:
                del self.active_tasks[bvid]
        else:
            if bvid in self.active_tasks:
                self.active_tasks[bvid]['status'] = "下载失败/中断"
                self.active_tasks[bvid]['progress'] = "ERROR"

    def scan_and_download(self, space_url="https://space.bilibili.com/2832224/video"):
        if not space_url.endswith("/video"):
            space_match = re.search(r'space\.bilibili\.com/(\d+)', space_url)
            if space_match:
                space_url = f"https://space.bilibili.com/{space_match.group(1)}/video"
                
        print(f"[{time.strftime('%Y-%m-%d %H:%M:%S')}] 开始巡视 B 站账号: {space_url}")
        
        # 1. 快速拿到最近的 10 个投稿信息 (此时拿不到标题也没关系)
        ydl_opts_flat = {'cookiefile': self.cookie_path, 'extract_flat': True, 'playlistend': 10, 'quiet': True}

        try:
            with yt_dlp.YoutubeDL(ydl_opts_flat) as ydl:
                info = ydl.extract_info(space_url, download=False)
                entries = list(info.get('entries', []))
        except Exception as e:
            print(f"❌ 扫描主页列表失败: {e}")
            return

        for entry in entries:
            # 2. 提取 BV 号
            bvid = entry.get('id', '')
            if not str(bvid).startswith('BV'):
                url = entry.get('url', '')
                bv_match = re.search(r'(BV\w+)', url)
                if bv_match:
                    bvid = bv_match.group(1)

            if not bvid: continue
            
            # 3. 过滤掉已经成功入库的视频
            if self._is_downloaded(bvid): continue

            # 4. 🚀 对于新视频，直接“模拟手动下载”去获取真实的详细信息！
            print(f"⏳ 发现新视频 {bvid}，正在获取详细元数据...")
            ydl_opts_full = {'cookiefile': self.cookie_path, 'quiet': True}
            
            with yt_dlp.YoutubeDL(ydl_opts_full) as ydl_full:
                try:
                    video_info = ydl_full.extract_info(f"https://www.bilibili.com/video/{bvid}", download=False)
                except Exception as e:
                    print(f"❌ 解析新视频失败 {bvid}: {e}")
                    continue
                    
            # 5. 拿到真实的标题，进行严格判定
            title = video_info.get('title', '')
            if not title: continue
            
            target_team = self._match_team(title)
            date_match = re.search(r'202\d{5}', title)
            
            # 🚀 如果不是目标队伍或者不是公演，直接跳过！不写历史记录！
            if not target_team or not date_match: 
                print(f"⏭️ [忽略] {title}")
                continue
            
            # 6. 一切条件符合，投入下载池
            date_str = date_match.group()
            year = date_str[:4]
            formatted_date = f"{year}-{date_str[4:6]}-{date_str[6:8]}"

            print(f"🎯 命中目标公演，准备下载: {title} ({bvid})")
            self.active_tasks[bvid] = {
                "title": title, "status": "解析视频分P...", "progress": "0%", "p_current": 0, "p_total": "?"
            }

            self._process_video_info(bvid, title, video_info, target_team, date_str, year, formatted_date)

    def download_single(self, video_url: str):
        # 手动下载的逻辑保持不变，因为它是绝对精准的
        print(f"[{time.strftime('%Y-%m-%d %H:%M:%S')}] 手动触发下载: {video_url}")
        bvid_match = re.search(r'(BV\w+)', video_url)
        if not bvid_match: return
        bvid = bvid_match.group(1)

        self.active_tasks[bvid] = {
            "title": f"手动解析: {bvid}", "status": "获取网络请求...", "progress": "0%", "p_current": 0, "p_total": "?"
        }

        ydl_opts_full = {'cookiefile': self.cookie_path, 'quiet': True}
        with yt_dlp.YoutubeDL(ydl_opts_full) as ydl_full:
            try:
                video_info = ydl_full.extract_info(f"https://www.bilibili.com/video/{bvid}", download=False)
            except Exception as e:
                self.active_tasks[bvid]['status'] = "解析源视频失败"
                return

            title = video_info.get('title', '')
            self.active_tasks[bvid]['title'] = title
            
            target_team = self._match_team(title)
            date_match = re.search(r'202\d{5}', title)
            
            if not target_team or not date_match:
                self.active_tasks[bvid]['status'] = "标题不符合入库规则"
                return
            
            date_str = date_match.group()
            year = date_str[:4]
            formatted_date = f"{year}-{date_str[4:6]}-{date_str[6:8]}"
            
            self._process_video_info(bvid, title, video_info, target_team, date_str, year, formatted_date)

    def scan_all(self):
        import yaml
        if not os.path.exists(self.config_path): return
        with open(self.config_path, 'r', encoding='utf-8') as f:
            data = yaml.safe_load(f)
        for sub in data.get('subscriptions', []):
            if sub.get('url'):
                self.scan_and_download(sub.get('url'))
```

import os
import yaml
from fastapi import FastAPI, BackgroundTasks
from fastapi.middleware.cors import CORSMiddleware
from apscheduler.schedulers.background import BackgroundScheduler
from scheduler import Snh48Scanner

app = FastAPI(title="SNH48 Downloader API")

# 允许跨域请求 (关键！允许前端 8090 端口访问后端 8095)
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

CONFIG_PATH = os.environ.get("CONFIG_PATH", "/config/roster.yaml")
COOKIE_PATH = os.environ.get("COOKIE_PATH", "/cookie/cookie.txt")
DOWNLOAD_PATH = os.environ.get("DOWNLOAD_PATH", "/downloads")

scanner = Snh48Scanner(CONFIG_PATH, COOKIE_PATH, DOWNLOAD_PATH)
scheduler = BackgroundScheduler()

@app.on_event("startup")
def start_scheduler():
    # 改为调用 scanner.scan_all
    scheduler.add_job(scanner.scan_all, 'cron', hour='13,23')
    scheduler.start()
    print("✅ 定时任务已启动")

@app.on_event("shutdown")
def stop_scheduler():
    scheduler.shutdown()

@app.get("/")
def read_root():
    return {"status": "running", "message": "SNH48 自动下载服务正在运行"}

@app.post("/api/trigger_scan")
def trigger_scan_manually(background_tasks: BackgroundTasks):
    background_tasks.add_task(scanner.scan_all)
    return {"message": "已在后台触发全局扫描任务"}

# ---------- 新增供前端调用的 API ----------

@app.get("/api/config")
def get_config():
    """获取花名册与订阅配置"""
    if os.path.exists(CONFIG_PATH):
        with open(CONFIG_PATH, 'r', encoding='utf-8') as f:
            return yaml.safe_load(f) or {}
    return {"subscriptions": [], "teams": []}

@app.post("/api/config")
def save_config(config: dict):
    """保存前端修改的配置并热重载"""
    with open(CONFIG_PATH, 'w', encoding='utf-8') as f:
        yaml.dump(config, f, allow_unicode=True, sort_keys=False)
    # 热重载后端的内存字典
    scanner.nfo_builder.roster = scanner.nfo_builder._load_roster(CONFIG_PATH)
    return {"message": "配置已保存并生效"}

@app.get("/api/history")
def get_history():
    """获取历史记录及当前活跃的下载任务"""
    history_file = scanner.history_file
    completed_lines = []
    if os.path.exists(history_file):
        with open(history_file, 'r') as f:
            # 倒序排列
            completed_lines = [line.strip() for line in f.readlines() if line.strip()][::-1]
            
    return {
        "completed": completed_lines,
        "active": scanner.active_tasks  # 将内存中的进度实时返回给前端
    }

def download_specific_video(payload: dict, background_tasks: BackgroundTasks):
    """处理前端传来的具体链接"""
    url = payload.get("url")
    if not url:
        return {"message": "未提供有效链接"}
    # 将下载任务放入后台队列，不阻塞网页响应
    background_tasks.add_task(scanner.download_single, url)
    return {"message": "指令已投递，系统正在后台解析下载"}

import os
import yt_dlp

class BiliDownloader:
    def __init__(self, cookie_path: str):
        self.cookie_path = cookie_path
        if not os.path.exists(self.cookie_path):
            print(f"警告: 未找到 Cookie 文件: {self.cookie_path}，下载高画质可能受限。")

    def download_part(self, bvid: str, p_number: int, output_path_without_ext: str, progress_callback=None) -> bool:
        ydl_opts = {
            'cookiefile': self.cookie_path,
            'outtmpl': f'{output_path_without_ext}.%(ext)s',
            'format': 'bestvideo+bestaudio/best',
            'merge_output_format': 'mp4',
            
            # 🚀 抛弃 Aria2，回归 yt-dlp 原生的 DASH 切片多线程并发（这就是真正的 IDM 模式）
            'concurrent_fragment_downloads': 10, # 开 10 个线程切片并发
            'socket_timeout': 15,                # 增加 15 秒超时防假死
            
            'writesubtitles': True,
            'subtitleslangs': ['all'],
            'playlist_items': str(p_number),
            'quiet': True,
            'no_warnings': True,
        }

        # 挂载前端进度条的回调钩子
        if progress_callback:
            ydl_opts['progress_hooks'] = [progress_callback]

        target_url = f"https://www.bilibili.com/video/{bvid}"
        print(f"开始并发切片下载: {bvid} - P{p_number} -> {output_path_without_ext}")
        
        try:
            with yt_dlp.YoutubeDL(ydl_opts) as ydl:
                error_code = ydl.download([target_url])
                
                # 💡 物理文件兜底校验
                if os.path.exists(f"{output_path_without_ext}.mp4"):
                    return True
                return error_code == 0
                
        except Exception as e:
            # 即使发生异常抛错，只要视频文件已存在，依然算作成功
            if os.path.exists(f"{output_path_without_ext}.mp4"):
                return True
            print(f"❌ yt-dlp 发生致命异常: {e}")
            return False

## backend/app/nfo_builder.py

```python
import os
import re
import yaml
import xml.etree.ElementTree as ET
from xml.dom import minidom

class NfoBuilder:
    def __init__(self, config_path: str, base_download_path: str):
        self.base_download_path = base_download_path
        self.roster = self._load_roster(config_path)

    def _load_roster(self, config_path: str) -> dict:
        if not os.path.exists(config_path):
            print(f"警告：未找到花名册文件 {config_path}")
            return {}
        with open(config_path, 'r', encoding='utf-8') as f:
            data = yaml.safe_load(f)
        
        roster_dict = {}
        if data and 'teams' in data:
            for team in data['teams']:
                if team.get('enabled', False):
                    roster_dict[team['name']] = {
                        "members": team.get('members', []),
                        "keywords": team.get('keywords', [team['name']])
                    }
        return roster_dict

    def _prettify_xml(self, elem) -> str:
        rough_string = ET.tostring(elem, 'utf-8')
        reparsed = minidom.parseString(rough_string)
        return '\n'.join([line for line in reparsed.toprettyxml(indent="  ").split('\n') if line.strip()])

    def build_metadata(self, team_name: str, year: str, date_str: str, p_number: str, part_title: str, plot: str, perf_title: str):
        """
        :param perf_title: 完整的 B 站稿件标题，用于创建该场公演的独立文件夹
        """
        show_dir = os.path.join(self.base_download_path, team_name)
        season_dir = os.path.join(show_dir, f"Season {year}")
        os.makedirs(season_dir, exist_ok=True)

        tvshow_nfo_path = os.path.join(show_dir, "tvshow.nfo")
        if not os.path.exists(tvshow_nfo_path):
            tvshow = ET.Element("tvshow")
            ET.SubElement(tvshow, "title").text = f"SNH48 {team_name}"
            ET.SubElement(tvshow, "plot").text = f"SNH48 {team_name} 日常剧场公演录播"
            with open(tvshow_nfo_path, "w", encoding="utf-8") as f:
                f.write(self._prettify_xml(tvshow))

        season_nfo_path = os.path.join(season_dir, "season.nfo")
        if not os.path.exists(season_nfo_path):
            season = ET.Element("season")
            ET.SubElement(season, "title").text = f"{year} 年度公演"
            ET.SubElement(season, "seasonnumber").text = year
            with open(season_nfo_path, "w", encoding="utf-8") as f:
                f.write(self._prettify_xml(season))

        # 🚀 新增：清理标题中的特殊字符，创建单场公演的独立文件夹
        safe_perf_title = re.sub(r'[\\/*?:"<>|]', "", perf_title)
        perf_dir = os.path.join(season_dir, f"{date_str} {safe_perf_title}")
        os.makedirs(perf_dir, exist_ok=True)

        mmdd = date_str.replace("-", "")[4:8] 
        episode_num = f"{mmdd}{p_number.zfill(2)}"
        
        episode_filename = f"S{year}E{episode_num} - {part_title}"
        episode_nfo_path = os.path.join(perf_dir, f"{episode_filename}.nfo")
        
        ep_details = ET.Element("episodedetails")
        ET.SubElement(ep_details, "title").text = part_title
        ET.SubElement(ep_details, "season").text = year
        ET.SubElement(ep_details, "episode").text = episode_num
        ET.SubElement(ep_details, "aired").text = date_str
        ET.SubElement(ep_details, "plot").text = plot

        # 🚀 修复演员名单 Bug：正确读取字典中的 members 数组
        team_data = self.roster.get(team_name, {})
        members = team_data.get("members", [])
        for member in members:
            actor_tag = ET.SubElement(ep_details, "actor")
            ET.SubElement(actor_tag, "name").text = member
            ET.SubElement(actor_tag, "role").text = "Member"

        with open(episode_nfo_path, "w", encoding="utf-8") as f:
            f.write(self._prettify_xml(ep_details))
        
        # 将文件输出到专属文件夹内
        return os.path.join(perf_dir, episode_filename)
```

    <script>
        const { createApp, ref, onMounted, onUnmounted } = Vue;

        createApp({
            setup() {
                const API_BASE = "http://192.168.1.12:8095/api";
                
                const history = ref([]);
                const activeTasks = ref({});
                const config = ref({ subscriptions: [], teams: [] });
                const scanMessage = ref("自动调度：每日 13:00 / 23:00");
                const isScanning = ref(false);
                const manualUrl = ref("");
                const isManualDownloading = ref(false);
                let pollInterval = null;

                const fetchHistory = async () => {
                    try {
                        const res = await fetch(`${API_BASE}/history`);
                        const data = await res.json();
                        history.value = data.completed;
                        activeTasks.value = data.active || {};
                    } catch (e) { /* ignore polling errors */ }
                };

                const fetchConfig = async () => {
                    try {
                        const res = await fetch(`${API_BASE}/config`);
                        const data = await res.json();
                        if (!data.subscriptions) data.subscriptions = [];
                        if (!data.teams) data.teams = [];
                        data.teams.forEach(t => {
                            if (!t.keywords) t.keywords = [];
                            if (!t.members) t.members = [];
                            if (!t.oshis) t.oshis = [];
                        });
                        config.value = data;
                    } catch (e) { console.error(e); }
                };

                const triggerScan = async () => {
                    if(isScanning.value) return;
                    try {
                        isScanning.value = true;
                        scanMessage.value = "指令发送中...";
                        const res = await fetch(`${API_BASE}/trigger_scan`, { method: 'POST' });
                        const data = await res.json();
                        scanMessage.value = data.message;
                        setTimeout(() => { isScanning.value = false; }, 3000);
                    } catch (e) {
                        scanMessage.value = "请求失败";
                        isScanning.value = false;
                    }
                };

                const triggerManualDownload = async () => {
                    if (!manualUrl.value) return;
                    try {
                        isManualDownloading.value = true;
                        const res = await fetch(`${API_BASE}/download_specific`, {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json' },
                            body: JSON.stringify({ url: manualUrl.value })
                        });
                        const data = await res.json();
                        alert('✅ ' + data.message);
                        manualUrl.value = ""; 
                    } catch (e) {
                        alert('❌ 请求失败，请检查后端状态');
                    } finally {
                        isManualDownloading.value = false;
                    }
                };

                const saveConfig = async () => {
                    try {
                        const res = await fetch(`${API_BASE}/config`, {
                            method: 'POST',
                            headers: { 'Content-Type': 'application/json' },
                            body: JSON.stringify(config.value)
                        });
                        const data = await res.json();
                        alert('✅ ' + data.message);
                    } catch (e) { alert('❌ 保存失败'); }
                };

                const addSub = () => config.value.subscriptions.push({ name: '', url: 'https://space.bilibili.com/...' });
                const removeSub = (idx) => { if(confirm('移除该订阅源？')) config.value.subscriptions.splice(idx, 1); };
                const addTeam = () => config.value.teams.push({ name: '新队伍', enabled: true, keywords: [], members: [], oshis: [] });
                const removeTeam = (idx) => { if(confirm('删除队伍？')) config.value.teams.splice(idx, 1); };

                const addKeyword = (team, event) => {
                    const val = event.target.value.trim();
                    if (val && !team.keywords.includes(val)) team.keywords.push(val);
                    event.target.value = '';
                };

                const addMember = (team, event) => {
                    const val = event.target.value.trim();
                    if (val && !team.members.includes(val)) team.members.push(val);
                    event.target.value = '';
                };

                const toggleOshi = (team, member) => {
                    if (!team.oshis) team.oshis = [];
                    const idx = team.oshis.indexOf(member);
                    if (idx > -1) team.oshis.splice(idx, 1);
                    else team.oshis.push(member);
                };

                onMounted(() => {
                    fetchHistory();
                    fetchConfig();
                    // 开启轮询，每 1.5 秒抓取一次后端状态，实现实时进度条
                    pollInterval = setInterval(fetchHistory, 1500);
                });

                onUnmounted(() => {
                    if(pollInterval) clearInterval(pollInterval);
                });

                return { 
                    history, activeTasks, config, scanMessage, isScanning, manualUrl, isManualDownloading,
                    fetchHistory, triggerScan, triggerManualDownload, saveConfig,
                    addSub, removeSub, addTeam, removeTeam, addKeyword, addMember, toggleOshi
                };
            }
        }).mount('#app')
    </script>
</body>
</html>
